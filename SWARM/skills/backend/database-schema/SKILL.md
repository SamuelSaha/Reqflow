## 📝 Normalization

### Normal Forms

```typescript
// ✅ 1NF: Atomic values, no repeating groups
// ❌ BAD — Multiple values in one column
{ userId: 1, phoneNumbers: '555-1234, 555-5678' }

// ✅ GOOD — Separate rows
{ userId: 1, phoneNumber: '555-1234' }
{ userId: 1, phoneNumber: '555-5678' }

// ✅ 2NF: No partial dependencies (all non-key attributes depend on full PK)
// ❌ BAD — Composite key with partial dependency
{ orderId: 1, productId: 5, productName: 'Shoes', quantity: 2 }
// productName depends only on productId, not full PK (orderId, productId)

// ✅ GOOD — Split into tables
// OrderItems: { orderId, productId, quantity }
// Products: { productId, name }

// ✅ 3NF: No transitive dependencies
// ❌ BAD — Non-key attribute depends on another non-key attribute
{ userId: 1, city: 'SF', zipCode: '94102', cityPopulation: 883305 }
// cityPopulation depends on city, not userId

// ✅ GOOD — Separate table
// Cities: { city, zipCode, population }
// Users: { userId, city }
```

### When to Denormalize

```typescript
// ✅ Read-heavy, write-rarely scenarios
const denormalizationPatterns = {
  // Counter caching
  user: {
    id: 'user-123',
    orderCount: 15, // Denormalized counter
    // vs querying COUNT(*) on every read
  },
  
  // Aggregated data
  orderSummary: {
    orderId: 'order-456',
    totalAmount: 150.00,
    itemCount: 3,
    // Pre-calculated to avoid joins
  },
  
  // Search optimization
  productSearch: {
    productId: 'prod-789',
    searchText: 'nike running shoes mens size 10', // Concatenated for full-text search
    categoryPath: 'sports > running > shoes', // Materialized path
  },
}
```

## 🔑 Primary Keys

### PK Strategies

```sql
-- ✅ Auto-increment (Simple, sequential)
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  email VARCHAR(255) NOT NULL
);

-- ✅ UUID (Distributed systems, no collision)
CREATE TABLE orders (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_id UUID NOT NULL,
  created_at TIMESTAMP DEFAULT NOW()
);

-- ✅ Snowflake ID (Distributed, sortable)
-- 41 bits timestamp | 10 bits machine | 12 bits sequence
CREATE TABLE events (
  id BIGINT PRIMARY KEY, -- Generated by Snowflake algorithm
  event_type VARCHAR(50),
  payload JSONB
);

-- ✅ Composite (Natural business keys)
CREATE TABLE order_items (
  order_id UUID,
  product_id UUID,
  quantity INT NOT NULL,
  PRIMARY KEY (order_id, product_id)
);
```

## 📊 Indexing Strategy

### Index Types

```sql
-- ✅ B-Tree (Default, good for equality and range queries)
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_orders_created_at ON orders(created_at);

-- ✅ Partial Index (Filter condition)
CREATE INDEX idx_active_users ON users(email) WHERE status = 'active';

-- ✅ Composite Index (Multi-column, order matters)
CREATE INDEX idx_orders_user_status ON orders(user_id, status, created_at DESC);
-- Use: WHERE user_id = ? AND status = ? ORDER BY created_at DESC

-- ✅ Unique Index (Enforce uniqueness + fast lookup)
CREATE UNIQUE INDEX idx_users_email_unique ON users(email) WHERE deleted_at IS NULL;

-- ✅ GIN Index (JSONB, arrays, full-text)
CREATE INDEX idx_products_metadata ON products USING GIN (metadata);

-- ✅ GiST Index (Geospatial, ranges)
CREATE INDEX idx_locations_coordinates ON locations USING GiST (coordinates);
```

### Index Best Practices


## 🔒 Constraints

### Data Integrity

```sql
-- ✅ NOT NULL - Required fields
CREATE TABLE users (
  id UUID PRIMARY KEY,
  email VARCHAR(255) NOT NULL,
  name VARCHAR(255) NOT NULL
);

-- ✅ UNIQUE - No duplicates
CREATE TABLE products (
  id UUID PRIMARY KEY,
  sku VARCHAR(100) UNIQUE NOT NULL,
  name VARCHAR(255) NOT NULL
);

-- ✅ CHECK - Validation rules
CREATE TABLE orders (
  id UUID PRIMARY KEY,
  total DECIMAL(10,2) CHECK (total >= 0),
  status VARCHAR(20) CHECK (status IN ('pending', 'paid', 'shipped', 'cancelled'))
);

-- ✅ FOREIGN KEY - Referential integrity
CREATE TABLE order_items (
  id UUID PRIMARY KEY,
  order_id UUID NOT NULL REFERENCES orders(id) ON DELETE CASCADE,
  product_id UUID NOT NULL REFERENCES products(id),
  quantity INT CHECK (quantity > 0)
);

-- ON DELETE behaviors:
-- CASCADE: Delete child records
-- SET NULL: Set FK to NULL
-- RESTRICT: Prevent deletion
-- NO ACTION: Same as RESTRICT (deferred)
```

## 🚀 Performance Optimization

### Query Optimization

```sql
-- ✅ EXPLAIN ANALYZE - Understand query plans
EXPLAIN ANALYZE
SELECT u.name, COUNT(o.id) as order_count
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
WHERE u.created_at > '2026-01-01'
GROUP BY u.id, u.name
ORDER BY order_count DESC;

-- Look for:
-- - Seq Scan vs Index Scan
-- - Nested Loop vs Hash Join
-- - High execution time
-- - High row counts
```

### Partitioning

```sql
-- ✅ Range Partitioning (Time-series data)
CREATE TABLE events (
  id UUID,
  event_type VARCHAR(50),
  created_at TIMESTAMP,
  payload JSONB
) PARTITION BY RANGE (created_at);

-- Create partitions
CREATE TABLE events_2026_01 PARTITION OF events
  FOR VALUES FROM ('2026-01-01') TO ('2026-02-01');

CREATE TABLE events_2026_02 PARTITION OF events
  FOR VALUES FROM ('2026-02-01') TO ('2026-03-01');

-- Automatic partition creation
-- Use pg_partman extension
```

## 🔄 Schema Migrations

### Migration Best Practices

```typescript
// ✅ Versioned migrations
// migrations/202602020001_add_user_status.ts
export async function up(knex: Knex): Promise<void> {
  await knex.schema.table('users', table => {
    table.enum('status', ['active', 'inactive', 'suspended'])
      .defaultTo('active')
      .notNullable()
    
    // Add index for new column
    table.index('status')
  })
}

export async function down(knex: Knex): Promise<void> {
  await knex.schema.table('users', table => {
    table.dropIndex('status')
    table.dropColumn('status')
  })
}
```

### Zero-Downtime Migrations

```typescript
// ✅ Pattern: Expand-contract for breaking changes

// Phase 1: Add new column (nullable)
await knex.schema.table('users', table => {
  table.string('email_normalized').nullable()
})

// Phase 2: Backfill data in batches
const batchSize = 1000
let offset = 0
while (true) {
  const users = await knex('users')
    .select('id', 'email')
    .limit(batchSize)
    .offset(offset)
  
  if (users.length === 0) break
  
  for (const user of users) {
    await knex('users')
      .where('id', user.id)
      .update({ email_normalized: normalizeEmail(user.email) })
  }
  
  offset += batchSize
}

// Phase 3: Make column NOT NULL
await knex.schema.table('users', table => {
  table.string('email_normalized').notNullable().alter()
})

// Phase 4: Update application code
// Phase 5: Drop old column (after code deployed)
```
